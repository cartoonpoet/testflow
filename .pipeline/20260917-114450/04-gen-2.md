import { z } from "zod";

/* ── 1. ActionType ───────────────────────────────────────── */

export const ACTION_TYPES = [
  "goto", "click", "fill", "select", "check", "uncheck",
  "press", "hover", "assert_visible", "assert_text", "assert_url", "wait",
] as const;

export const ActionTypeSchema = z.enum(ACTION_TYPES);
export type ActionType = z.infer<typeof ActionTypeSchema>;

/** 시안 스텝 카드의 `<code>` 칩과의 대응표. web 에서 재정의 금지. */
export const ACTION_CHIP_LABEL = {
  goto: "이동",   click: "클릭",  fill: "입력",   select: "입력",
  check: "입력",  uncheck: "입력", press: "입력",  hover: "클릭",
  assert_visible: "확인", assert_text: "확인", assert_url: "확인", wait: "대기",
} as const satisfies Record<ActionType, string>;

/** 인스펙터 "동작" select 3종 (문구는 01-clarify 시안 그대로). */
export const ASSERTION_ACTION_LABEL = {
  assert_visible: "화면에 표시되는지 확인",
  assert_text: "텍스트 값 확인",
  assert_url: "URL 확인",
} as const;
export type AssertionActionType = keyof typeof ASSERTION_ACTION_LABEL;

/* ── 2. Locator — 단일 값이 아니라 "순위 배열" (FR-004) ───── */

export const LOCATOR_BY = ["role", "label", "text", "testid", "css"] as const;
export const LocatorBySchema = z.enum(LOCATOR_BY);
export type LocatorBy = z.infer<typeof LocatorBySchema>;

/** 후보가 여러 개 매칭될 때만 쓰는 0-based 순번. 없으면 "유일해야 한다"는 뜻. */
const nthField = { nth: z.number().int().nonnegative().optional() };

export const RoleLocatorSchema = z.object({
  by: z.literal("role"),
  role: z.string().min(1),
  name: z.string().optional(),
  exact: z.boolean().optional(),
  ...nthField,
});
export const LabelLocatorSchema = z.object({
  by: z.literal("label"), value: z.string().min(1), exact: z.boolean().optional(), ...nthField,
});
export const TextLocatorSchema = z.object({
  by: z.literal("text"), value: z.string().min(1), exact: z.boolean().optional(), ...nthField,
});
export const TestIdLocatorSchema = z.object({
  by: z.literal("testid"), value: z.string().min(1), ...nthField,
});
/** ★ 고급 설정 전용 — 테스터 화면에 절대 노출하지 않는다. */
export const CssLocatorSchema = z.object({
  by: z.literal("css"), value: z.string().min(1), ...nthField,
});

export const LocatorCandidateSchema = z.discriminatedUnion("by", [
  RoleLocatorSchema, LabelLocatorSchema, TextLocatorSchema, TestIdLocatorSchema, CssLocatorSchema,
]);
export type LocatorCandidate = z.infer<typeof LocatorCandidateSchema>;

/** css 를 제외한 후보 — 테스터에게 보여도 되는 것들. */
export const PublicLocatorCandidateSchema = z.discriminatedUnion("by", [
  RoleLocatorSchema, LabelLocatorSchema, TextLocatorSchema, TestIdLocatorSchema,
]);
export type PublicLocatorCandidate = z.infer<typeof PublicLocatorCandidateSchema>;

/** 실패 진단용 스냅샷. 재생에는 쓰지 않는다. */
export const LocatorSnapshotSchema = z.object({
  tag: z.string().min(1),
  attrs: z.record(z.string(), z.string()).optional(),
  text: z.string().optional(),
});
export type LocatorSnapshot = z.infer<typeof LocatorSnapshotSchema>;

/** `test_steps.target_json` 의 형태. */
export const LocatorTargetSchema = z.object({
  primary: LocatorCandidateSchema,
  fallbacks: z.array(LocatorCandidateSchema).default([]),
  frameUrl: z.string().nullable().optional(),
  snapshot: LocatorSnapshotSchema.nullable().optional(),
});
export type LocatorTarget = z.infer<typeof LocatorTargetSchema>;

/** API 응답용 — css 가 전부 제거된 형태. primary 가 css 였으면 null 이 된다. */
export const PublicLocatorTargetSchema = z.object({
  primary: PublicLocatorCandidateSchema.nullable(),
  fallbacks: z.array(PublicLocatorCandidateSchema),
  frameUrl: z.string().nullable().optional(),
  snapshot: LocatorSnapshotSchema.nullable().optional(),
});
export type PublicLocatorTarget = z.infer<typeof PublicLocatorTargetSchema>;

const isPublicCandidate = (c: LocatorCandidate): c is PublicLocatorCandidate => c.by !== "css";

/** API 응답 직렬화 직전에 반드시 통과시킨다 (Task 4.7: 응답에 "css" 문자열이 없어야 한다). */
export function toPublicLocatorTarget(target: LocatorTarget): PublicLocatorTarget {
  const primary = isPublicCandidate(target.primary) ? target.primary : null;
  return {
    primary,
    fallbacks: target.fallbacks.filter(isPublicCandidate),
    frameUrl: target.frameUrl ?? null,
    snapshot: target.snapshot ?? null,
  };
}

/** primary + fallbacks 를 시도 순서대로 편 배열. runner 해석 진입점. */
export function locatorChain(target: LocatorTarget): LocatorCandidate[] {
  return [target.primary, ...target.fallbacks];
}

/* ── 3. 입력값 / 옵션 ────────────────────────────────────── */

/**
 * `test_steps.input_json`.
 *   fill/select/press → 입력할 값 · goto → 이동 URL · assert_text/assert_url → 기대값
 * isSecret:true 면 화면에 `••••••••`, DB 에는 `{{password}}` 같은 참조만 남는다.
 */
export const TestStepInputSchema = z.object({
  value: z.string().max(2000),
  isSecret: z.boolean().default(false),
});
export type TestStepInput = z.infer<typeof TestStepInputSchema>;

/** 인스펙터 "최대 대기 시간" select 선택지 (시안: 5초 · 10초 · 30초). */
export const STEP_TIMEOUT_CHOICES_MS = [5000, 10000, 30000] as const;
export const DEFAULT_STEP_TIMEOUT_MS = 10000;

export const TestStepOptionsSchema = z.object({
  timeoutMs: z.number().int().min(100).max(300_000).default(DEFAULT_STEP_TIMEOUT_MS),
  /** true 면 실패해도 계속 진행한다(스텝은 `skipped`). */
  optional: z.boolean().default(false),
  /** `action_type: 'wait'` 전용. */
  waitMs: z.number().int().positive().max(300_000).optional(),
});
export type TestStepOptions = z.infer<typeof TestStepOptionsSchema>;

export const DEFAULT_STEP_OPTIONS: TestStepOptions = {
  timeoutMs: DEFAULT_STEP_TIMEOUT_MS,
  optional: false,
};

/* ── 4. TestStep ─────────────────────────────────────────── */

const ACTIONS_REQUIRING_TARGET = new Set<ActionType>([
  "click", "fill", "select", "check", "uncheck", "press", "hover",
  "assert_visible", "assert_text",
]);
const ACTIONS_REQUIRING_INPUT = new Set<ActionType>([
  "goto", "fill", "select", "press", "assert_text", "assert_url",
]);

/** 확장 가능한 원형(ZodObject). `.extend()`/`.omit()` 이 필요하면 이쪽. */
export const TestStepBaseSchema = z.object({
  id: z.uuid().optional(),           // 서버가 부여. 신규/녹화초안엔 없다
  scenarioId: z.uuid().optional(),
  sequence: z.number().int().positive(),   // 1부터. 재정렬 시 전량 재기입
  name: z.string().min(1).max(200),        // 업무 단계 이름
  actionType: ActionTypeSchema,
  target: LocatorTargetSchema.nullable().optional(),
  input: TestStepInputSchema.nullable().optional(),
  options: TestStepOptionsSchema.default(DEFAULT_STEP_OPTIONS),
});

function checkStepShape(step: z.infer<typeof TestStepBaseSchema>, ctx: z.RefinementCtx): void {
  const { actionType, target, input, options } = step;
  if (ACTIONS_REQUIRING_TARGET.has(actionType) && !target)
    ctx.addIssue({ code: "custom", path: ["target"], message: `'${actionType}' 동작에는 대상(target)이 필요합니다.` });
  if (ACTIONS_REQUIRING_INPUT.has(actionType) && !input)
    ctx.addIssue({ code: "custom", path: ["input"], message: `'${actionType}' 동작에는 값(input)이 필요합니다.` });
  if (actionType === "goto" && target)
    ctx.addIssue({ code: "custom", path: ["target"], message: "'goto' 동작에는 대상(target)을 지정할 수 없습니다." });
  if (actionType === "wait" && options.waitMs === undefined)
    ctx.addIssue({ code: "custom", path: ["options", "waitMs"], message: "'wait' 동작에는 options.waitMs 가 필요합니다." });
}

/** ★ 단일 타입 소스. web·api·runner 는 이 스키마만 쓴다. */
export const TestStepSchema = TestStepBaseSchema.superRefine(checkStepShape);

export type TestStep = z.infer<typeof TestStepSchema>;          // 파싱 결과(기본값 채워짐)
export type TestStepPayload = z.input<typeof TestStepSchema>;    // 요청 body 타입

/** 녹화 초안 — sequence 는 시나리오 반영 시 부여하므로 아직 없다. */
export const DraftStepSchema = TestStepBaseSchema
  .omit({ id: true, scenarioId: true, sequence: true })
  .superRefine((draft, ctx) => { checkStepShape({ ...draft, sequence: 1 }, ctx); });
export type DraftStep = z.infer<typeof DraftStepSchema>;

/** 스텝 배열 — sequence 가 1..n 으로 연속인지까지 본다. `PUT /steps` 검증용. */
export const TestStepArraySchema = z.array(TestStepSchema).superRefine((steps, ctx) => {
  steps.forEach((step, index) => {
    if (step.sequence !== index + 1)
      ctx.addIssue({ code: "custom", path: [index, "sequence"],
        message: `sequence 는 1부터 연속이어야 합니다 (기대: ${String(index + 1)}, 실제: ${String(step.sequence)}).` });
  });
});

/** 화면 표시용 마스킹 문자열 (시안 `••••••••`). */
export const SECRET_MASK = "••••••••";

export function displayStepValue(step: Pick<TestStep, "input">): string {
  if (!step.input) return "";
  return step.input.isSecret ? SECRET_MASK : step.input.value;
}

### ⚠️ 판별 필드 이름에 대한 결정 — `kind` 가 아니라 `by` 다

오케스트레이터 지시문은 후보 형태를 `{kind: 'role'|'label'|...}` 로 적었으나 **`by` 로 확정했다.**
근거 3가지 —

1. 02-context 의 `test_steps.target_json` 주석 블록이 `{"by":"role", ...}` 로 적혀 있다(정본 DDL).
2. 03-phases **Task 2.1** 의 작업 설명이 `by: role|label|text|testid|css` 다.
3. 03-phases **Task 4.7** 의 완료기준이 "`target_json.fallbacks` 의 `by:'css'` 항목을 기본 제외"로
   다운스트림 Task 에서 이미 `by` 를 전제하고 있다.

지시문의 `kind` 는 형태를 서술한 산문으로 판단했다. **바꿔야 한다면 지금이 마지막 기회다**
(Gen-Phase 3 이후로 넘어가면 web·api·runner 3곳 + 마이그레이션 주석을 함께 고쳐야 한다).

### 엔티티 ↔ DDL 정합성 스모크 상세

실제 MySQL 에 TypeORM 리포지토리로 쓰고 읽어 확인했다(마이그레이션 실행만으로는 컬럼명 매핑이
맞는지 증명되지 않는다).

```
target_json 왕복: {"primary":{"by":"role","name":"로그인","role":"button","exact":true},
                   "fallbacks":[{"by":"css","value":"form > button.submit"}]}
options_json 왕복: {"optional":false,"timeoutMs":10000}
zod 재검증: true
draft_steps 왕복 건수: 1
스텝 CASCADE 삭제됨: true          ← 시나리오 삭제 시 test_steps CASCADE
run.scenario_id SET NULL: true     ← 시나리오 삭제돼도 실행 이력 보존
SMOKE OK
```

9개 엔티티 전부(projects/scenarios/test_steps/suites/suite_scenarios/runs/step_results/
artifacts/recording_sessions)에 INSERT·SELECT 를 돌렸고 JSON 컬럼 3종의 왕복, 한글 저장,
`ON DELETE CASCADE` / `SET NULL` 동작을 확인했다.

### 계획과 다르게 한 것 / 사실 기록

- **`yarn test` 는 Gen-Phase 1 시점에 실패하던 상태였다.** `apps/{api,runner,web}` 에 테스트
  파일이 없어 vitest 가 exit 1 을 냈다(Gen-Phase 1 검증 로그에 `yarn test` 항목이 없다).
  세 워크스페이스의 `test` 스크립트에 `--passWithNoTests` 를 붙여 해소했다.
- **`migration:revert` 를 CLI 스크립트로 만들었다.** TypeORM 1.1.1 CLI(`typeorm migration:run -d …`)
  를 쓰지 않고 `runMigrations()`/`undoLastMigration()` 을 직접 호출하는 `dist/cli/*.js` 를 뒀다.
  04-gen-1 의 "컴파일된 JS 로 돌린다" 지시를 가장 단순하게 만족하는 형태다.
- **엔티티 클래스 이름에 `Entity` 접미사를 붙였다** (`ProjectEntity`, `TestStepEntity` …).
  `@testflow/contracts` 가 `Project`/`Scenario`/`TestStep`/`Run`/`StepResult`/`Artifact`/
  `Suite`/`RecordingSession` 타입을 **전부 같은 이름으로** 내보내므로, 접미사 없이는 두 패키지를
  동시에 import 하는 api/runner 에서 이름이 충돌한다.
- **`typeorm_migrations` 테이블 이름을 명시했다.** Gen-Phase 1 스파이크가 기본값 `migrations`
  테이블을 DB 에 남겨 뒀다(빈 테이블). 혼동을 막으려 이름을 분리하고 스파이크 잔재는 DROP 했다.

## 이슈 / 결정사항

### 1. Locator 판별 필드는 `by` 로 확정했다 ★

위 "판별 필드 이름에 대한 결정" 절 참조. 지시문의 `kind` 대신 02-context DDL 주석과
03-phases Task 2.1/4.7 의 표기를 따랐다. **되돌리려면 지금 지시하라.**

### 2. `runs` 에 `variables` 컬럼을 아예 만들지 않았다

지시는 "`variables` 를 저장하려면 Secret 을 `'***'` 로 치환하거나 컬럼에서 제외하라"였다.
**컬럼 제외**를 택했다 — 컬럼이 존재하면 언젠가 누군가 평문을 넣는다. 나중에 "어떤 키를 썼는가"를
남길 필요가 생기면 `maskVariablesForStorage()` 를 거친 `variables_masked JSON NULL` 을
새 마이그레이션으로 추가하는 편이 안전하다(코드 주석에도 남겼다).

### 3. `superRefine` 때문에 스키마를 2단으로 나눴다

zod 의 `.superRefine()` 결과에는 `.extend()`/`.omit()`/`.partial()` 을 쓸 수 없다.
그런데 `CreateStepDto`·`PatchStepDto`·`DraftStep` 이 전부 `TestStep` 에서 파생돼야 한다.
→ `TestStepBaseSchema`(순수 ZodObject, 파생용) + `TestStepSchema`(교차검증 포함, 최종)로 분리했다.
**새 DTO 를 만들 때는 `TestStepBaseSchema` 에서 파생시켜라.**

### 4. `artifacts.size_bytes` 는 TS 에서 `string` 이다

MySQL `BIGINT UNSIGNED` + `bigNumberStrings: true` 조합이라 드라이버가 문자열로 돌려준다.
JS number 정밀도(2^53) 를 넘길 일은 없지만, 드라이버 기본 동작을 바꾸면 다른 BIGINT 컬럼까지
영향받으므로 그대로 뒀다. API 응답 스키마(`ArtifactSchema.sizeBytes`)는 `number` 이므로
**서비스 레이어에서 `Number()` 변환이 필요하다.**

### 5. contracts 는 node 타입에 의존하지 않는다

`apps/web`(브라우저)에서도 import 되므로 tsconfig 에서 `types: []` 로 두고 `Buffer` /
`NodeJS.ReadableStream` 을 쓰지 않았다. `StorageAdapter.getStream()` 의 반환 타입은
`AsyncIterable<Uint8Array>` 다(Node 의 `Readable` 이 이 타입을 만족한다).

### 6. 범위 준수

- Gen-Phase 3 이후 Task 는 **구현하지 않았다.** `apps/runner/src/record/` 디렉토리 자체가 없다
  (`screencast.ts` · `input-bridge.ts` 미생성).
- 다른 레포는 **읽지도 수정하지도 않았다.**
- `docker-compose.yml` 의 `TESTFLOW_` 접두사 변수는 **손대지 않았다.**

- `git commit` 까지만 했고 **push 하지 않았다**(원격 없음).
